% =====================================================================
%  config/biblio.tex
%  Bibliographie avec biblatex + biber (compatible Unicode/multilingue).
%
%  STYLE : APA 7e édition (auteur-date).
%  ---------------------------------------------------------------------
%  La FTI n'impose PAS un style unique : l'aide-mémoire des mémoires de
%  maîtrise (MA traduction / MATECH) précise que « les citations et
%  références suivent les consignes de votre unité ou de votre jury »
%  et que la présentation formelle est fixée « par votre directeur ou
%  directrice de mémoire ». Le format attendu est de type AUTEUR-DATE.
%
%  Pour le Département TIM (technologies de la traduction, TAL), le choix
%  par défaut retenu ici est APA 7 (auteur-date), qui est aussi la norme
%  de référence à l'échelle de l'UNIGE. Il se charge via « biblatex-apa »
%  (présent sur Overleaf et dans une installation TeX Live complète).
%
%  ROBUSTESSE : si « biblatex-apa » n'est pas installé (sous-ensemble
%  TeX Live minimal), on retombe automatiquement sur le style natif
%  « authoryear », lui aussi auteur-date. Le document compile donc partout.
% =====================================================================

\usepackage{csquotes} % requis par biblatex ; sûr même si déjà chargé

\IfFileExists{apa7.bbx}{%
  % ---- Style APA 7 (préféré) ----------------------------------------
  \usepackage[
    backend=biber,
    style=apa,            % APA 7e édition (biblatex-apa)
    sortlocale=fr_FR,
    language=french,
    apamaxprtauth=99,     % liste tous les auteurs dans la référence
  ]{biblatex}
  % Localisation française des mentions APA (« et al. », « & », etc.).
  \DeclareLanguageMapping{french}{french-apa}
}{%
  % ---- Repli : style auteur-date natif ------------------------------
  \usepackage[
    backend=biber,
    style=authoryear,
    sorting=nyt,
    sortlocale=fr_FR,
    language=french,
    autolang=none,
    maxbibnames=99,
    maxcitenames=2,
    giveninits=true,
    uniquename=init,
    doi=true, isbn=false, url=true,
  ]{biblatex}
  \PackageWarning{config/biblio}{%
    biblatex-apa introuvable : repli sur le style « authoryear ».
    Sur Overleaf / TeX Live complet, APA 7 sera utilisé automatiquement.%
  }
}

% Fichier de références.
\addbibresource{biblio/references.bib}
